<script>
// ── State ──────────────────────────────────────────────────────────────────
const projects = {};
let eventSource = null;

// ── Tab switching ──────────────────────────────────────────────────────────
function switchTab(name, btn) {
  document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));
  document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));
  document.getElementById('panel-' + name).classList.add('active');
  btn.classList.add('active');
  if (name === 'projects') renderProjects();
  if (name === 'status') loadStatus();
}

function switchLab(name, btn) {
  document.querySelectorAll('.lab-section').forEach(s => s.classList.remove('active'));
  document.querySelectorAll('.lab-tab').forEach(b => b.classList.remove('active'));
  document.getElementById('lab-' + name).classList.add('active');
  btn.classList.add('active');
}

// ── Chat output ────────────────────────────────────────────────────────────
function addMsg(type, text, targetId) {
  const output = document.getElementById(targetId || 'chatOutput');
  const div = document.createElement('div');
  div.className = 'msg ' + type;
  div.textContent = text;
  output.appendChild(div);
  output.scrollTop = output.scrollHeight;
}

// ── Build ──────────────────────────────────────────────────────────────────
async function handleSend() {
  const input = document.getElementById('cmdInput');
  const idea = input.value.trim();
  if (!idea) return;
  input.value = '';

  addMsg('user', idea);
  addMsg('system', 'Building brief...');

  try {
    const res = await fetch('/api/build', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ idea })
    });
    const data = await res.json();

    if (data.error) { addMsg('error', 'Error: ' + data.error); return; }

    const pid = data.project_id;
    projects[pid] = { id: pid, idea, brief: data.brief, status: 'pending_approval', results: [] };

    addMsg('assistant', 'Brief generated: ' + (data.brief?.summary || 'Ready'));
    addMsg('system', 'Project ID: ' + pid);

    const pidEl = document.createElement('div');
    pidEl.className = 'project-id';
    pidEl.textContent = '[' + pid + '] → approve?';
    pidEl.onclick = () => approveProject(pid);
    document.getElementById('chatOutput').appendChild(pidEl);
    document.getElementById('chatOutput').scrollTop = 99999;

  } catch (e) {
    addMsg('error', 'Request failed: ' + e.message);
  }
}

// ── Approve ────────────────────────────────────────────────────────────────
async function approveProject(pid) {
  addMsg('system', 'Approving ' + pid + '...');
  if (projects[pid]) projects[pid].status = 'executing';

  try {
    const res = await fetch('/api/approve', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ project_id: pid })
    });
    const data = await res.json();

    if (data.job_id && !data.job_id.startsWith('direct:')) {
      addMsg('assistant', 'Queued → job ' + data.job_id);
      addMsg('system', 'Watch BOARDROOM tab for live updates');
      connectEventStream();
    } else {
      addMsg('assistant', 'Execution complete');
      if (projects[pid]) { projects[pid].status = 'completed'; projects[pid].results = data.results || []; }
    }
  } catch (e) {
    addMsg('error', 'Approve failed: ' + e.message);
  }
}

// ── Quick Build ────────────────────────────────────────────────────────────
async function quickBuild() {
  const prompt = document.getElementById('quickInput').value.trim();
  if (!prompt) return;
  addMsg('system', 'Running...', 'quickOutput');

  try {
    const res = await fetch('/api/quick', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ prompt })
    });
    const data = await res.json();
    addMsg('assistant', data.output || JSON.stringify(data), 'quickOutput');
  } catch (e) {
    addMsg('error', 'Failed: ' + e.message, 'quickOutput');
  }
}

// ── Job Status ─────────────────────────────────────────────────────────────
async function checkJob() {
  const jobId = document.getElementById('jobInput').value.trim();
  if (!jobId) return;

  try {
    const res = await fetch('/api/job/' + jobId + '/status');
    const data = await res.json();
    addMsg('system', JSON.stringify(data, null, 2), 'jobOutput');
  } catch (e) {
    addMsg('error', 'Failed: ' + e.message, 'jobOutput');
  }
}

// ── Event Stream ───────────────────────────────────────────────────────────
function connectEventStream() {
  if (eventSource) eventSource.close();
  const feed = document.getElementById('eventFeed');
  feed.innerHTML = '';

  eventSource = new EventSource('/api/events');

  eventSource.onmessage = (e) => {
    try {
      const evt = JSON.parse(e.data);
      if (evt.event_type === 'HEARTBEAT') return;

      const line = document.createElement('div');
      line.className = 'evt-line';
      const t = new Date(evt.created_at || Date.now()).toLocaleTimeString();
      line.innerHTML =
        '<span class="evt-time">' + t + '</span>' +
        '<span class="evt-type">' + evt.event_type + '</span>' +
        '<span class="evt-payload">' + JSON.stringify(evt.payload || {}) + '</span>';
      feed.appendChild(line);
      feed.scrollTop = feed.scrollHeight;

      // Mirror to home chat
      addMsg('event', evt.event_type + ' ' + JSON.stringify(evt.payload || {}));

    } catch(err) {}
  };

  eventSource.onerror = () => {
    const line = document.createElement('div');
    line.className = 'evt-line';
    line.innerHTML = '<span class="evt-type" style="color:#ff4d6d">DISCONNECTED</span>';
    feed.appendChild(line);
  };
}

// ── Projects ───────────────────────────────────────────────────────────────
function renderProjects() {
  const list = document.getElementById('projectsList');
  const keys = Object.keys(projects);

  if (!keys.length) {
    list.innerHTML = '<div class="no-projects">No projects yet. Build something first.</div>';
    return;
  }

  list.innerHTML = '';
  keys.reverse().forEach(pid => {
    const p = projects[pid];
    const card = document.createElement('div');
    card.className = 'project-card';
    card.innerHTML = `
      <div class="project-card-header">
        <span class="project-card-id">${pid}</span>
        <span class="project-card-status status-${p.status}">${p.status.toUpperCase()}</span>
      </div>
      <div class="project-card-idea">${p.idea}</div>
      <div class="project-card-brief">${p.brief?.summary || ''}</div>
      ${p.status === 'pending_approval' ? `<button class="approve-btn" onclick="approveProject('${pid}');switchTab('home',document.querySelector('.nav-btn'))">APPROVE &rarr; EXECUTE</button>` : ''}
      ${p.results?.length ? `<div class="project-output open" id="out-${pid}">${p.results.map(r => '[' + r.module + ']\n' + r.output).join('\n\n---\n\n')}</div>` : ''}
    `;
    list.appendChild(card);
  });
}

// ── Status ─────────────────────────────────────────────────────────────────
async function loadStatus() {
  const grid = document.getElementById('statusGrid');
  grid.innerHTML = '<div class="status-card"><div class="status-card-name">LOADING</div><div class="status-card-val status-unk">...</div></div>';

  try {
    const res = await fetch('/api/status');
    const data = await res.json();
    grid.innerHTML = '';

    Object.entries(data).forEach(([key, val]) => {
      const ok = val === true || val === 'ok' || val === 200;
      const card = document.createElement('div');
      card.className = 'status-card';
      card.innerHTML = `
        <div class="status-card-name">${key.toUpperCase()}</div>
        <div class="status-card-val ${ok ? 'status-ok' : typeof val === 'string' && val.includes('err') ? 'status-err' : 'status-unk'}">${JSON.stringify(val)}</div>
      `;
      grid.appendChild(card);
    });
  } catch (e) {
    grid.innerHTML = '<div class="status-card"><div class="status-card-name">ERROR</div><div class="status-card-val status-err">' + e.message + '</div></div>';
  }
}

// ── Keyboard shortcuts ─────────────────────────────────────────────────────
document.addEventListener('keydown', e => {
  if (e.key === 'Enter' && document.activeElement === document.getElementById('cmdInput')) handleSend();
  if (e.key >= '1' && e.key <= '4' && e.target.tagName !== 'INPUT' && e.target.tagName !== 'TEXTAREA') {
    const btns = document.querySelectorAll('.nav-btn');
    const tabs = ['home','projects','lab','status'];
    const idx = parseInt(e.key) - 1;
    if (btns[idx]) switchTab(tabs[idx], btns[idx]);
  }
});
